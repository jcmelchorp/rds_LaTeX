\subsubsection{Potencias con números negativos}

     \ejemplosboxed[6]{Realiza las siguientes potencias de números negativos:
     
      \begin{multicols}{3}
            \begin{parts}\large
                  \part $-7^2=$ \fillin[$-49$][0in]
                  \part $-(-2)^4=$ \fillin[$-16$][0in]
                  \part $(-3)^3=$ \fillin[$-27$][0in]
                  %\part $-(-5)^3=$ \fillin[$125$][0in]
            \end{parts}
      \end{multicols}
	  }
