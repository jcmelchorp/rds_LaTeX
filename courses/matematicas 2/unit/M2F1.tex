			\begin{parts}\footnotesize%
				\part Calcula la altura de un prisma que tiene como área de la base 6 m$^2$ y 99 m$^3$ de capacidad.

				\begin{solutionbox}{2cm}

				\end{solutionbox}

                            \begin{align*}
                                -x-2 & = 15                \\
                                -x   & =15+2               \\
                                -x   & =17                 \\
                                x    & = \frac{17}{-1}=-17

                        \end{solutionbox}

                        \part $ 11x-33=55 $

                        \begin{solutionbox}{3.5cm}\footnotesize%
