Contesta las siguientes preguntas:

            \begin{multicols}{2}
                \begin{parts}
                    \part Las calificaciones de un salón de secundaria son las siguientes: 80, 82, 85, 88, 90, 88, 91, 85, 95, 88, 88, 97, 100. ¿Cuál es la \textbf{mediana} de las calificaciones?
                    \fillin[88][2cm]

                    \begin{solutionbox}{2cm}
                        Ordenando los datos se obtiene:\\[-0.2em]
                        $\left\{80, 82, 85, 85, 88, 88, 88, 88, 90, 91, 95, 97, 100 \right\} $\\[-0.2em]
                        $\therefore$ Mediana es 88
                    \end{solutionbox}

                    \part Las edades de un grupo de personas son: 44, 41, 47, 48, 44, 39, 45, 49, 44 y 47 años. ¿Cuál es la \textbf{mediana} de las edades?
                    \fillin[44.5][2cm]

                    \begin{solutionbox}{2cm}
                        Ordenando los datos se obtiene: \\[-0.2em]
                        $\left\{39, 41, 44, 44, 44, 45, 47, 47, 48, 49 \right\} $ \\[-0.2em]
                        $\therefore$ Mediana es 44.5
                    \end{solutionbox}

                \end{parts}
            \end{multicols}
