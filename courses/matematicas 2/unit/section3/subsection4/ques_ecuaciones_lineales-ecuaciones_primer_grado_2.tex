Resuelve las siguientes ecuaciones:

        \begin{multicols}{2}
            \begin{parts}
                \part $\large -3(2x-5)=-1$

                \begin{solutionbox}{3.2cm}\footnotesize%
                    \begin{align*}
                        -3(2x-5) & =-1                           \\
                        -6x+15   & =-1                           \\
                        -6x      & =-1-15                        \\
                        -6x      & =-16                          \\
                        x        & =\sfrac{-16}{-6}=\sfrac{8}{3}
                    \end{align*}
                \end{solutionbox}

                \part $\large -4(3x+5)=5(-2x-3)$

                \begin{solutionbox}{3.2cm}\footnotesize%
                    \begin{align*}
                        -4(3x+5) & =5(-2x-3)                     \\
                        -12x-20  & =-10x-15                      \\
                        -12x+10x & =-15+20                       \\
                        -2x      & =5                            \\
                        x        & =-\sfrac{5}{-2}=-\sfrac{5}{2}
                    \end{align*}
                \end{solutionbox}
            \end{parts}
        \end{multicols}
