Encuentra el \textit{n-ésimo} término de la siguientes sucesiones aritméticas:
                \begin{multicols}{2}
                    \begin{parts}
                        \part Calcula el término número 28 de la siguiente sucesión aritmética: $-69,-72,-75,-78,-81,\ldots$
                        \begin{solutionbox}{1.5cm}
                            \[-3(28)-66=-84-66=-150\]
                        \end{solutionbox}

                        \part Calcula el término número 47 de la siguiente sucesión aritmética: $-5,0,5,10,15,\ldots$
                        \begin{solutionbox}{1.5cm}
                            \[5(47)-5=235-5=225\]
                        \end{solutionbox}
                    \end{parts}
                \end{multicols}
