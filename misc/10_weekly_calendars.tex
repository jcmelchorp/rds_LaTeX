% Command to print the activities for a specific day and week
\newcommand{\printday}[2]{%
  \begin{tcolorbox}[title={\Large\textbf{#1}}, colback=white, colframe=teal!70!black, height=4.25cm, boxrule=1pt, arc=4pt]
    % Iterate through the database and filter by week and day
    \DTLforeach{activities}{\MyWeek=Week,\MyDay=Day,\MyTime=Time,\MyActivity=Activity}{%
      \ifthenelse{\equal{\MyWeek}{#2} \and \equal{\MyDay}{#1}}{%
        \noindent{\color{teal!80!black}\textbullet}\hspace{0.1cm}\textbf{\MyTime}\\[0.05cm]
        \MyActivity \\[0.3cm]
      }{}%
    }%
  \end{tcolorbox}
}
