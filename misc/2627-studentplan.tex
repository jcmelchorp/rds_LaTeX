\newboolean{isholiday}
\newcommand{\checkholiday}[2]{% #1 = mes (MM), #2 = día (DD)
  \setboolean{isholiday}{false}%
  \def\holidayname{}%
  
  % --- VACACIONES (Por Rangos) ---
  % Vacaciones de Invierno (21 Dic - 6 Ene)
  \ifthenelse{\number#1=12 \and \number#2>20}{\setboolean{isholiday}{true}\def\holidayname{Vacaciones de Invierno}}{}%
  \ifthenelse{\number#1=1 \and \number#2<7}{\setboolean{isholiday}{true}\def\holidayname{Vacaciones de Invierno}}{}%
  
  % Vacaciones de Primavera (22 Mar - 2 Abr)
  \ifthenelse{\number#1=3 \and \number#2>21}{\setboolean{isholiday}{true}\def\holidayname{Vacaciones de Primavera}}{}%
  \ifthenelse{\number#1=4 \and \number#2<3}{\setboolean{isholiday}{true}\def\holidayname{Vacaciones de Primavera}}{}%

  % --- FERIADOS FIJOS Y CTE ---
  % (Se colocan después para que sobreescriban el nombre genérico de "Vacaciones" 
  % si coinciden exactamente en ese día, ej. Navidad y Año Nuevo)
  
  \ifthenelse{\equal{#1}{01} \and \equal{#2}{01}}{\setboolean{isholiday}{true}\def\holidayname{Año Nuevo}}{}%
  \ifthenelse{\equal{#1}{05} \and \equal{#2}{01}}{\setboolean{isholiday}{true}\def\holidayname{Día del Trabajo}}{}%
  \ifthenelse{\equal{#1}{05} \and \equal{#2}{05}}{\setboolean{isholiday}{true}\def\holidayname{Batalla de Puebla}}{}%
  \ifthenelse{\equal{#1}{09} \and \equal{#2}{16}}{\setboolean{isholiday}{true}\def\holidayname{Independencia}}{}%
  \ifthenelse{\equal{#1}{11} \and \equal{#2}{02}}{\setboolean{isholiday}{true}\def\holidayname{Día de Muertos}}{}%
  \ifthenelse{\equal{#1}{12} \and \equal{#2}{12}}{\setboolean{isholiday}{true}\def\holidayname{Día de la Virgen}}{}%
  \ifthenelse{\equal{#1}{12} \and \equal{#2}{25}}{\setboolean{isholiday}{true}\def\holidayname{Navidad}}{}%
  
  % Asuetos móviles y CTE (Con formato 01, 02 para evitar fallos de lectura)
  \ifthenelse{\equal{#1}{11} \and \equal{#2}{16}}{\setboolean{isholiday}{true}\def\holidayname{Revolución (Asueto)}}{}%
  \ifthenelse{\equal{#1}{09} \and \equal{#2}{25}}{\setboolean{isholiday}{true}\def\holidayname{CTE Consejo Técnico Escolar}}{}%
  \ifthenelse{\equal{#1}{10} \and \equal{#2}{30}}{\setboolean{isholiday}{true}\def\holidayname{CTE Consejo Técnico Escolar}}{}%
  \ifthenelse{\equal{#1}{11} \and \equal{#2}{27}}{\setboolean{isholiday}{true}\def\holidayname{CTE Consejo Técnico Escolar}}{}%
  \ifthenelse{\equal{#1}{01} \and \equal{#2}{29}}{\setboolean{isholiday}{true}\def\holidayname{CTE Consejo Técnico Escolar}}{}%
  \ifthenelse{\equal{#1}{02} \and \equal{#2}{26}}{\setboolean{isholiday}{true}\def\holidayname{CTE Consejo Técnico Escolar}}{}%
  \ifthenelse{\equal{#1}{04} \and \equal{#2}{30}}{\setboolean{isholiday}{true}\def\holidayname{CTE Consejo Técnico Escolar}}{}%
  \ifthenelse{\equal{#1}{05} \and \equal{#2}{28}}{\setboolean{isholiday}{true}\def\holidayname{CTE Consejo Técnico Escolar}}{}%
  \ifthenelse{\equal{#1}{06} \and \equal{#2}{25}}{\setboolean{isholiday}{true}\def\holidayname{CTE Consejo Técnico Escolar}}{}%
}

% Comando para imprimir las actividades de un día y semana específicos
\newcommand{\printday}[4]{%
  % #1 = Nombre del día a mostrar (ej. Lunes)
  % #2 = Número de semana (\w)
  % #3 = Desplazamiento de días (0=Lunes, 1=Martes, etc.)
  % #4 = Nombre del día en tu archivo CSV (ej. Lunes o Monday)
  
  % Calcular la fecha exacta sumando semanas y días a la fecha de inicio
  \newcount\currentdate
  \newcount\weekoffset
  \weekoffset=#2\relax
  \advance\weekoffset by -1\relax
  \multiply\weekoffset by 7\relax
  \advance\weekoffset by #3\relax
  \currentdate=\startdate
  \advance\currentdate by \weekoffset
  \pgfcalendarjuliantodate{\currentdate}{\myyear}{\mymonth}{\myday}%
  
  \begin{tcolorbox}[title={\textbf{#1 \myday} \footnotesize \mesespanol{\mymonth}}, halign title=center, colback=white, colframe=teal!70!white, height=3.8cm, boxrule=1pt, arc=2pt]
    
    % Revisar si la fecha calculada es un día feriado o vacacional
    \checkholiday{\mymonth}{\myday}%
    
    \ifthenelse{\boolean{isholiday}}{%
      % --- SI ES FERIADO/VACACIONES: Imprimir cartel y omitir actividades ---
      \vspace{0.3cm}
      \begin{center}
        \textcolor{gray!80!black}{\textbf{\holidayname}}\\[0.1cm]
        \textcolor{gray!80!black}{\textit{(Sin clases)}}
      \end{center}
    }{%
      % --- SI NO ES FERIADO: Iterar por la base de datos y filtrar ---
      \DTLforeach{activities}{\MyWeek=Week,\MyDay=Day,\MyTime=Time,\MyActivity=Activity}{%
        \ifthenelse{\equal{\MyWeek}{#2} \and \equal{\MyDay}{#4}}{%
          \small\textbf{\MyTime} \MyActivity \\[0.3cm]
        }{}%
      }%
    }%
  \end{tcolorbox}%
}
