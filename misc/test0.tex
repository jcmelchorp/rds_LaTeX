\NewDocumentCommand{\studentlist}{m}
{%
  \studentlist:n { #1 }
}

\tl_new:N \studentlist_header

% Función auxiliar para iterar con \DTLforeach e inyectar fechas en la tabla
\cs_new_protected:Nn \append_dates_from_db:Nn
  {
    % #1 = variable token list del header (ej. \studentlist_header_sep)
    % #2 = nombre de la base de datos de datatool (ej. fechasSep)
    \DTLforeach{#2}{\mydate=Fecha, \mycolor=Color}
      {
        \tl_put_right:Nx #1
          {
            & \exp_not:N \rotatebox{90}
              {
                \tl_if_blank:VF \mycolor { \exp_not:N \cellcolor{\mycolor} }
                \exp_not:N \DTMdate{\mydate}
              }
          }
      }
    \tl_put_right:Nn #1 { \\ \hline }
  }

\cs_new_protected:Nn \studentlist:n
{
  \seq_clear:N \studentlist_seq
  \ior_open:Nn \studentlist_stream { #1.csv } 
  \int_zero:N \l_tmpa_int
  
  \ior_map_inline:Nn \studentlist_stream
  {
    \int_incr:N \l_tmpa_int
    \seq_put_right:Nx \studentlist_seq
    {
      \int_to_arabic:n { \l_tmpa_int } & \exp_not:n { ##1 } &
    }
  }

  % =========================================================
  % SEPTIEMBRE
  % =========================================================
  \tl_set:Nn \studentlist_header_sep
  {
    \multicolumn{2}{r}{\fontsize{14}{8}\selectfont  \leveltitle{#1}} &
    \multicolumn{1}{|c|}{\cellcolor{Goldenrod!50}\fontsize{10}{8}\selectfont \bfseries A.}  & 
    \multicolumn{22}{|c|}{\cellcolor{RoyalBlue!50}\fontsize{14}{12}\selectfont \bfseries Septiembre}  & 
    \multicolumn{2}{|c|}{\cellcolor{ForestGreen!50}\fontsize{10}{8}\selectfont \bfseries Oct.} \\
    & & \multicolumn{5}{|c|}{Semana~1} & \multicolumn{5}{|c|}{Semana~2} & \multicolumn{5}{|c|}{Semana~3} &
    \multicolumn{5}{|c|}{Semana~4} & \multicolumn{5}{|c|}{Semana~5} \\ \cline{3-27}
    \# & \textbf{Nombre~del~alumno} 
  }
  % Genera las 25 columnas dinámicamente:
  \append_dates_from_db:Nn \studentlist_header_sep { fechasSep }

  % =========================================================
  % OCTUBRE
  % =========================================================
  \tl_set:Nn \studentlist_header_oct
  {
    \multicolumn{2}{r}{\fontsize{14}{8}\selectfont  \leveltitle{#1}} &    
    \multicolumn{20}{c}{\cellcolor{ForestGreen!50}\fontsize{14}{12}\selectfont \bfseries Octubre} & 
    \multicolumn{5}{|c|}{\cellcolor{Brown!50}\fontsize{12}{10}\selectfont \bfseries Noviembre} \\
    & & \multicolumn{5}{|c|}{Semana~6} & \multicolumn{5}{|c|}{Semana~7} & \multicolumn{5}{|c|}{Semana~8} &
    \multicolumn{5}{|c|}{Semana~9} & \multicolumn{5}{|c|}{Semana~10} \\ \cline{3-27}
    \# & \textbf{Nombre~del~alumno} 
  }
  \append_dates_from_db:Nn \studentlist_header_oct { fechasOct }

  % === SE OMITEN LOS DEMÁS MESES POR BREVEDAD, EL PATRÓN ES IDÉNTICO ===
  % Sustituye el bloque \tl_set de Noviembre, Enero, etc. y añade su respectivo \append_dates_from_db:Nn 

  % Revisión de partición
  \int_compare:nT { \seq_count:N \studentlist_seq > 25 }
  {
    \__studentlist_halve:
  }
  
  \par\noindent
  \begin{tcolorbox}[listabox]
  \begin{tabular}{rw{l}{16em}|*{25}{w{c}{0.8em}|}}
    \tl_use:N \studentlist_header_sep
    \seq_map_function:NN \studentlist_seq \__studentlist_row:n \\
  \end{tabular}
  \end{tcolorbox}
  \newpage

  \par\noindent
  \begin{tcolorbox}[listabox]
  \begin{tabular}{rw{l}{16em}|*{25}{w{c}{0.8em}|}}
    \tl_use:N \studentlist_header_oct
    \seq_map_function:NN \studentlist_seq \__studentlist_row:n \\
  \end{tabular}
  \end{tcolorbox}
  % Añadir las demás cajas (Nov, Ene...)
}

\cs_new_protected:Nn \__studentlist_halve:
{
  \int_if_odd:nT { \seq_count:N \studentlist_seq }
  {
    \seq_put_right:Nn \studentlist_seq { \multicolumn{25}{c}{} }
  }
  \seq_clear:N \studentlist_temp_seq
  \int_step_inline:nn { \seq_count:N \studentlist_seq / 2 }
  {
    \seq_put_right:Nx \studentlist_temp_seq
    {
      \seq_item:Nn \studentlist_seq { ##1 } &
      \seq_item:Nn \studentlist_seq { ##1 + \seq_count:N \studentlist_seq / 2 }
    }
  }
  \seq_set_eq:NN \studentlist_seq \studentlist_temp_seq
  \tl_put_right:Nx \studentlist_header { & \exp_not:V \studentlist_header }
}

\cs_new_protected:Nn \__studentlist_row:n
{
  #1 & & & & & & & & & & & & & & & & & & & & & & & & \\ \hline
}
